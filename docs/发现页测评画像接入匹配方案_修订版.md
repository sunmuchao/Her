## 二、问题诊断（五问法）

```
问题现象：发现页匹配没有使用心理测评数据
├─ 为什么 1: evaluate_candidate() 没有读取 self_personality_traits_json
│   → search_matching.py 只处理 profile 字段，未关联 persona 表

├─ 为什么 3: 候选人返回数据没有 personality 字段
│   → service_integrations 和 view_models 未增加 personality 注入
├─ 为什么 4: persona_row 加载后未传递 personality 数据
```

**根本对策**: 建立 persona 测评数据 → search_matching → discovery_agent → 前端的完整数据桥梁

## 五、评分公式设计

### 5.1 依恋风格适配分（修复版）

```python
# partner_search/personality_matching.py

def calculate_attachment_fit(
    user_attachment: dict,
    candidate_attachment: dict,
) -> tuple[float, list[str], list[str]]:
    """
    基于依恋理论的风险分级模型
    
    参考: Mikulincer & Shaver (2016) 《Attachment in Adulthood》
    
    返回: (score, matched_on, risk_flags)
    """
    u_anx = user_attachment.get("anxiety", 0) / 100  # 转换为 0-1 范围
    u_avo = user_attachment.get("avoidance", 0) / 100
    c_anx = candidate_attachment.get("anxiety", 0) / 100
    c_avo = candidate_attachment.get("avoidance", 0) / 100
    
    matched_on = []
    risk_flags = []
    
    # 1. 双方都安全型：高分（最稳定组合）
    if u_anx < 0.3 and u_avo < 0.3 and c_anx < 0.3 and c_avo < 0.3:
        matched_on.append("双方都是安全型，关系节奏稳定")
        return 0.9, matched_on, risk_flags
    
    # 2. 一方安全型 + 另一方非极端：中等分
    if (u_anx < 0.3 and u_avo < 0.3) or (c_anx < 0.3 and c_avo < 0.3):
        if max(u_anx, u_avo, c_anx, c_avo) < 0.7:
            matched_on.append("一方安全型可稳住另一方")
            return 0.6, matched_on, risk_flags
    
    # 3. 焦虑型 + 回避型：低分（追逐-逃避模式）
    if u_anx > 0.5 and c_avo > 0.5:
        risk_flags.append("焦虑+回避组合，可能形成追逐-逃避循环")
        return 0.2, matched_on, risk_flags
    if c_anx > 0.5 and u_avo > 0.5:
        risk_flags.append("焦虑+回避组合，可能形成追逐-逃避循环")
        return 0.2, matched_on, risk_flags
    
    # 4. 双方相似的非安全型：中等分
    if abs(u_anx - c_anx) < 0.2 and abs(u_avo - c_avo) < 0.2:
        matched_on.append("依恋倾向相似，相处节奏更一致")
        return 0.5, matched_on, risk_flags
    
    return 0.4, matched_on, risk_flags
```

### 5.2 大五人格适配分（差异化权重）

```python
def calculate_big_five_fit(
    user_big_five: dict,
    candidate_big_five: dict,
) -> tuple[float, list[str], list[str]]:
    """
    基于 Luo & Klohnen (2005) 研究的加权模型
    
    核心发现：
    - 神经质差异对关系稳定性影响最大
    - 外向性差异影响最小（可以互补）
    """
    u_scores = user_big_five.get("scores", {})
    c_scores = candidate_big_five.get("scores", {})
    
    if not u_scores or not c_scores:
        return None, [], []
    
    # 不同维度的权重（基于实证研究）
    weights = {
        "neuroticism": 0.30,     # 神经质差异影响最大
        "agreeableness": 0.20,   # 宜人性差异影响较大
        "conscientiousness": 0.20,
        "openness": 0.15,
        "extraversion": 0.15,    # 外向性差异影响最小（可互补）
    }
    
    matched_on = []
    total_diff = 0
    
    for dim, weight in weights.items():
        u_val = u_scores.get(dim, 50)
        c_val = c_scores.get(dim, 50)
        diff = abs(u_val - c_val) / 100
        total_diff += diff * weight
        
        # 生成匹配原因
        if diff < 0.15:
            matched_on.append(f"{dim}维度相似")
    
    score = 1.0 - total_diff
    return score, matched_on, []
```

### 5.3 价值观适配分

```python
def calculate_values_fit(
    user_values: dict,
    candidate_values: dict,
) -> tuple[float, list[str], list[str]]:
    """
    基于 Byrne (1971) 相似性-吸引理论
    
    改进: 考虑价值观顺序权重 + 冲突检测
    """
    user_top = user_values.get("top_values", [])
    candidate_top = candidate_values.get("top_values", [])
    
    if not user_top or not candidate_top:
        return None, [], []
    
    matched_on = []
    risk_flags = []
    
    user_set = set(user_top)
    candidate_set = set(candidate_top)
    overlap = user_set & candidate_set
    
    if overlap:
        matched_on.append(f"价值观有重叠: {list(overlap)[:3]}")
    
    # 基础重叠分
    base_score = len(overlap) / max(len(user_top), len(candidate_top))
    
    # 冲突检测（可选）
    conflict_pairs = [
        ("完全自由的生活", "稳定可预期的日子"),
        ("冒险和挑战", "安稳平淡"),
    ]
    
    for pair in conflict_pairs:
        if (pair[0] in user_set and pair[1] in candidate_set) or \
           (pair[1] in user_set and pair[0] in candidate_set):
            risk_flags.append(f"价值观有潜在冲突")
            base_score -= 0.1
    
    return max(0, base_score), matched_on, risk_flags
```

### 5.4 总分合成

```python
def calculate_personality_fit_score(
    user_traits: PersonalityTraitsContext,
    candidate_traits: PersonalityTraitsContext,
) -> PersonalityFitScore:
    """
    合成测评适配分
    
    关键原则：
    - 只有双方都有对应测评才计算
    - 缺失的子分不参与（不是 0 分）
    - None 表示数据不足，不参与排序
    """
    result = PersonalityFitScore()
    
    available_scores = []
    
    # 依恋适配
    if user_traits.availability.get("has_attachment") and \
       candidate_traits.availability.get("has_attachment"):
        score, matched, risks = calculate_attachment_fit(
            user_traits.attachment, candidate_traits.attachment
        )
        result.attachment_fit = score
        result.matched_on.extend(matched)
        result.risk_flags.extend(risks)
        available_scores.append(("attachment", score, 0.3))
    
    # 价值观适配
    if user_traits.availability.get("has_values") and \
       candidate_traits.availability.get("has_values"):
        score, matched, risks = calculate_values_fit(
            user_traits.values, candidate_traits.values
        )
        result.values_fit = score
        result.matched_on.extend(matched)
        result.risk_flags.extend(risks)
        available_scores.append(("values", score, 0.4))
    
    # 气质适配（优先大五，其次 MBTI）
    if user_traits.availability.get("has_big_five") and \
       candidate_traits.availability.get("has_big_five"):
        score, matched, risks = calculate_big_five_fit(
            user_traits.big_five, candidate_traits.big_five
        )
        if score is not None:
            result.temperament_fit = score
            result.matched_on.extend(matched)
            available_scores.append(("temperament", score, 0.3))
    
    # 计算总分（按权重合成）
    if available_scores:
        total_weight = sum(w for _, _, w in available_scores)
        result.total_score = sum(s * (w / total_weight) for _, s, w in available_scores)
    else:
        result.total_score = None  # 数据不足
    
    return result
```
